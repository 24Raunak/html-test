// 1) Function Declaration

  console.log('Hi');

Hello()

// 2) Function Expression
let Hello = function() {
  console.log('Hi');
}

Function Declaration
When JavaScript is executed, function declaration (in this case, Hello() ) will automatically come at the top
This behavior is called as JavaScript hoisting.

Array Method - Map, filter, reduce

Map :-

let a = [1, 2, 3, 4, 5, 6]
let b = a.map((each) => each * 2);
console.log(b);

Output :-
[2, 4, 6, 8, 10, 12]


where 'each' represents each element of array
and 'each * 2' is the condition

Filter :-

let a = [1, 2, 3, 4, 5, 6]
let b = a.filter((each) => each > 3);
console.log(b);

Output :-
[4, 5, 6]

Reduce :-

let a = [1, 2, 3, 4, 5];
let b = a.reduce((ac, cv) => ac + cv);
console.log(b);

Output :-

15

where, cv = current value, ac = accumulator
